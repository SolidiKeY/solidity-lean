/-! Guards omission of the actual HeadCheck certificate, representation equality,
capacity hypothesis or short-input bound from joint field witnesses. Boundary
controls distinguish a three-byte head from a complete selector while preserving
caller tail words; they do not certify native loading or full entry execution. -/
open Solidity Solidity.Backend Solidity.Backend.AbiEntry

def shortInput : Solidity.AbiEntry.Input := ⟨[0,0,0],0⟩
def completeInput : Solidity.AbiEntry.Input := ⟨[0,0,0,0],0⟩

theorem complete_head_flag :
    Scalar.run (HeadCheck.code []) ((4 : Scalar.Word)::[11,99]) =
      some (Scalar.flag false::[11,99]) :=
  HeadCheck.correct [] 4 [11,99]

theorem short_head_flag :
    Scalar.run (HeadCheck.code []) ((3 : Scalar.Word)::[11,99]) =
      some (Scalar.flag true::[11,99]) :=
  HeadCheck.correct [] 3 [11,99]

theorem complete_head_entry :
    Solidity.AbiEntry.entry completeInput 0 [] = .ok .nil := rfl

theorem reject_all_head_refused :
    Scalar.run (HeadCheck.code []) ((4 : Scalar.Word)::[11,99]) ≠
      some (Scalar.flag true::[11,99]) := by
  rw [complete_head_flag]
  decide

theorem revert_all_entry_refused :
    Solidity.AbiEntry.entry completeInput 0 [] ≠ .error .revert := by
  rw [complete_head_entry]
  intro impossible
  cases impossible
