open Solidity Solidity.Backend Solidity.Backend.Selection

def selectionOpcodeInput : Scalar.Stack := [23,7,99]

def selectionOpcodeMachine (input : Scalar.Stack) : Evm.Machine :=
  { (Evm.Machine.init (fun _ => 17) ⟨9, by decide⟩) with
    stack := irStack input
    store := Evm.WordStore.ofFn (fun _ => 29)
    mem := fun _ => .val ⟨31, by decide⟩
    bal₀ := fun _ => 13
    caller := ⟨3, by decide⟩
    callvalue := ⟨2, by decide⟩
    timestamp := ⟨11, by decide⟩
    accepts := fun _ _ => false }

theorem selection_opcode_stack_coupled :
    (selectionOpcodeMachine selectionOpcodeInput).stack = irStack selectionOpcodeInput := rfl

theorem selection_opcode_world_populated :
    (selectionOpcodeMachine selectionOpcodeInput).bal ⟨9, by decide⟩ = 17 ∧
    (selectionOpcodeMachine selectionOpcodeInput).bal₀ ⟨9, by decide⟩ = 13 ∧
    (selectionOpcodeMachine selectionOpcodeInput).caller.val = 3 ∧
    (selectionOpcodeMachine selectionOpcodeInput).callvalue.val = 2 ∧
    (selectionOpcodeMachine selectionOpcodeInput).timestamp.val = 11 ∧
    (selectionOpcodeMachine selectionOpcodeInput).accepts ⟨9, by decide⟩ ⟨2, by decide⟩ = false := by decide

theorem selection_opcode_bad_stack_refused :
    ¬ (selectionOpcodeMachine selectionOpcodeInput).stack = irStack [0] := by
  intro wrong
  have lengths : (3 : Nat) = 1 := congrArg List.length wrong
  cases lengths

theorem selection_add_fixture_executes :
    Evm.Instr.add.step (selectionOpcodeMachine selectionOpcodeInput) =
      .ok { selectionOpcodeMachine selectionOpcodeInput with stack := irStack [30,99] } 0 := by
  exact (addCertificate.success selectionOpcodeInput [30,99]
    (selectionOpcodeMachine selectionOpcodeInput) rfl).mpr (by decide)

theorem selection_add_underflow_refused :
    Evm.Instr.add.step (selectionOpcodeMachine [23]) = .fault := by
  exact (addCertificate.refusal [23] (selectionOpcodeMachine [23]) rfl).mpr (by decide)

/-! Guard dropped constructor proof fields and reversed V1/V2 capacity reasoning.
This adapter forgets V2 guarantees; it never upgrades a legacy certificate. -/
def selectionLegacyOfV2 {input output : Scalar.Stack}
    (certificate : V2.FrameCertificate input output) : FrameCertificate input output where
  code := certificate.code
  frame := certificate.frame
  prefixes := by
    intro tail bound before after reached split ran
    apply Scalar.prefix_capacity before after ran
    rw [split]
    exact bound

def selectionDupFrame : V2.FrameCertificate [7] [7,7] := V2.duplicateHead 7

def selectionLegacyDup : FrameCertificate [7] [7,7] := selectionLegacyOfV2 selectionDupFrame

theorem selection_add_component_selected : selectArithmetic .add = some addCertificate := rfl

theorem selection_dup_frame_executes :
    Scalar.run selectionDupFrame.code [7,99] = some [7,7,99] := selectionDupFrame.frame [99]

theorem selection_legacy_dup_executes :
    Scalar.run selectionLegacyDup.code [7,99] = some [7,7,99] := selectionLegacyDup.frame [99]

theorem selection_dup_boundary_fits :
    Scalar.stackBound selectionDupFrame.code
      ((([7] : Scalar.Stack) ++ List.replicate 1022 99).length : Int) = true := by
  apply selectionDupFrame.capacity (List.replicate 1022 99)
  simp only [List.length_append, List.length_cons, List.length_nil, List.length_replicate]
  decide

theorem selection_dup_overflow_refused :
    ¬ Nonempty (V2.FrameCertificate (([7] : Scalar.Stack) ++ List.replicate 1023 99)
      (([7,7] : Scalar.Stack) ++ List.replicate 1023 99)) := by
  apply V2.no_overfull_output
  simp only [List.length_append, List.length_cons, List.length_nil, List.length_replicate]
  decide
