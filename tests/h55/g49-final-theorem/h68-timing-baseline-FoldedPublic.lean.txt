import Solidity.Roadmap.G49Final.PassCorrectAdapters

/-! H64/H65 folded G49/G50 statement over frozen G01 interfaces.
The actual certified passes form an intrinsic list. Only its fold establishes
public refinement. Every other unproved public/deployment/session obligation
is an explicitly named typed field, not an assumed PublicCorrectness oracle.
This conditional theorem does not construct missing producers or discharge
the residual fields, source admission, concrete V08 migration or lane coverage. -/
namespace Solidity.Roadmap.G49Final

/-- One public map, independent of input, program, artifact or bound. -/
def sourceEnv {S : SourceSpec} {I : IntrinsicSpec} {Asm : Lang Event Outcome} {V : EvmSpec}
    (lowering : Lowering S I) (backend : Compilation I Asm) (execution : Execution Asm V)
    (e : V.lang.Env) : S.lang.Env :=
  lowering.pass.envAbs (backend.pass.envAbs (execution.pass.envAbs e))

/-- The export uses exactly the same Lowering and composed intrinsic map. -/
def foldedExport {S : SourceSpec} {I : IntrinsicSpec} {Asm : Lang Event Outcome} {V : EvmSpec}
    (lowering : Lowering S I) (backend : Compilation I Asm) (execution : Execution Asm V)
    (decode : List Bytes → V.World → Option I.World) : StateExport S I V where
  lowering := lowering
  decode := decode
  envAbs e := backend.pass.envAbs (execution.pass.envAbs e)

/-- Source well-formedness/profile are supplied once, at the source end.
Later adapters use actual total-domain proofs, without another caller premise. -/
def pipeline {S : SourceSpec} {I : IntrinsicSpec} {Asm : Lang Event Outcome} {V : EvmSpec}
    (lowering : Lowering S I) (backend : Compilation I Asm) (execution : Execution Asm V)
    (source : SourceProgram S) (art : Artifact)
    (artifact : execution.assemble (backend.compile
      (lowering.lower source.resolved source.wellFormed source.inProfile art)) = art) :
    PassList (Behaviour.ofLang S.lang
      (S.meaning source.resolved source.wellFormed source.inProfile art))
      (Behaviour.ofLang V.lang (V.load art)) :=
  .cons (lowering_adapter lowering source.resolved source.wellFormed source.inProfile art)
    (.cons (compilation_adapter backend
      (lowering.lower source.resolved source.wellFormed source.inProfile art))
      (.cons (execution_adapter execution _ art artifact) (.nil _)))

/-- Program-indexed folds retain the SAME public environment abstraction. -/
theorem pipeline_env {S : SourceSpec} {I : IntrinsicSpec} {Asm : Lang Event Outcome} {V : EvmSpec}
    (lowering : Lowering S I) (backend : Compilation I Asm) (execution : Execution Asm V)
    (source : SourceProgram S) (art : Artifact)
    (artifact : execution.assemble (backend.compile
      (lowering.lower source.resolved source.wellFormed source.inProfile art)) = art)
    (e : V.lang.Env) :
    (pipeline lowering backend execution source art artifact).fold.envAbs e =
      sourceEnv lowering backend execution e := rfl

/-- All remaining frozen public obligations, unchanged in meaning. These are
open typed premises; none is inferred from the existence of the pass template. -/
structure PublicObligations {S : SourceSpec} {I : IntrinsicSpec}
    {Asm : Lang Event Outcome} {V : EvmSpec}
    (lowering : Lowering S I) (backend : Compilation I Asm) (execution : Execution Asm V)
    (compiler : S.Input → Option Artifact) (decode : List Bytes → V.World → Option I.World) : Prop where
  sourceApproved : S.Approved
  sourceNonVacuous : S.NonVacuous
  targetApproved : V.Approved
  keccak : V.fixed.Keccak
  coverage : ∀ i, (∃ r, S.Admitted i r) ↔ (compiler i).isSome
  bothPipelines : S.AllPipelines
  artifact : ∀ i r, S.Admitted i r → ∀ (w : S.WellFormed r) (p : S.InProfile r) art,
    compiler i = some art → execution.assemble (backend.compile (lowering.lower r w p art)) = art
  fixed : S.fixed = V.fixed
  session : ∀ e, S.session (sourceEnv lowering backend execution e) = V.session e
  ifixed : I.fixed = V.fixed
  isession : ∀ e, I.session (backend.pass.envAbs (execution.pass.envAbs e)) = V.session e
  creates_table : ∀ i art, compiler i = some art → art.PrefixFree ∧
    ∀ e rt, V.created (V.load art) e rt =
      (createsOf (V.lang.sem (V.load art) e rt).events).map fun ca => art.initcode ca.1 ++ ca.2
  hashes_source : ∀ i r, S.Admitted i r → ∀ (w : S.WellFormed r) (p : S.InProfile r) art,
    compiler i = some art → ∀ e rt, V.lang.Allowed e → V.lang.Admissible rt →
      (∀ b, b ∈ V.hashed (V.load art) e rt →
        ∃ rs, S.lang.Admissible rs ∧ b ∈ S.hashed (S.meaning r w p art)
          (sourceEnv lowering backend execution e) rs) ∧
      ∀ b, b ∈ V.slotHashed (V.load art) e rt →
        ∃ rs, S.lang.Admissible rs ∧ b ∈ S.slotHashed (S.meaning r w p art)
          (sourceEnv lowering backend execution e) rs
  decode_reach : ∀ i r, S.Admitted i r → ∀ (w : S.WellFormed r) (p : S.InProfile r) art,
    compiler i = some art → ∀ e, V.lang.Allowed e → ∀ rt, V.lang.Admissible rt →
      V.lang.Guard (V.load art) e rt → ∃ ri, I.lang.Admissible ri ∧
        (V.worlds (V.load art) e rt).map (decode (V.hashed (V.load art) e rt)) =
          (I.worlds (I.ofWorld (lowering.lower r w p art))
            (backend.pass.envAbs (execution.pass.envAbs e)) ri).map some
  decode_commutes : ∀ i r, S.Admitted i r → ∀ (_ : S.WellFormed r) (_ : S.InProfile r) art,
    compiler i = some art → ∀ e rt, V.lang.Allowed e → V.lang.Admissible rt →
      V.lang.Guard (V.load art) e rt → ∀ ω, ω ∈ V.worlds (V.load art) e rt →
      ∀ ω', decode (V.hashed (V.load art) e rt) ω = some ω' →
      ∀ e', V.lang.Allowed e' → (V.session e').balance = V.balances ω →
        I.lang.Allowed (backend.pass.envAbs (execution.pass.envAbs e')) ∧
        RefinesUnder I.lang.Admissible V.lang.Admissible
          (I.lang.Forward (I.ofWorld ω') (backend.pass.envAbs (execution.pass.envAbs e')))
          (I.lang.Guard (I.ofWorld ω') (backend.pass.envAbs (execution.pass.envAbs e')))
          (fun rt' => HashPremise V.fixed
            (V.hashed (V.load art) e rt ++ V.hashed (V.resume ω) e' rt')
            (V.slotHashed (V.load art) e rt ++ V.slotHashed (V.resume ω) e' rt'))
          (I.lang.sem (I.ofWorld ω') (backend.pass.envAbs (execution.pass.envAbs e')))
          (V.lang.sem (V.resume ω) e')

/-- The frozen G49/G50 public theorem is assembled from the pass-list fold
and the explicitly retained residual obligations. The fold supplies complete
refinement and environment surjectivity. Deployment, resumed-session and
hash/source facts remain named premises, not weakened or silently discharged. -/
theorem g49_g50_of_fold {S : SourceSpec} {I : IntrinsicSpec}
    {Asm : Lang Event Outcome} {V : EvmSpec}
    (lowering : Lowering S I) (backend : Compilation I Asm) (execution : Execution Asm V)
    (compiler : S.Input → Option Artifact) (decode : List Bytes → V.World → Option I.World)
    (obligations : PublicObligations lowering backend execution compiler decode) :
    S.Approved ∧ S.NonVacuous ∧ V.Approved ∧ V.fixed.Keccak ∧
      PublicCorrectness S V compiler (sourceEnv lowering backend execution)
        (foldedExport lowering backend execution decode) ∧
      SpecTransfer S V compiler (sourceEnv lowering backend execution) := by
  have publicProof : PublicCorrectness S V compiler (sourceEnv lowering backend execution)
      (foldedExport lowering backend execution decode) := {
    coverage := obligations.coverage
    bothPipelines := obligations.bothPipelines
    envOnto := by
      intro e allowed
      obtain ⟨i, r, w, p, _, _, _, admitted, _⟩ := obligations.sourceNonVacuous.1
      have compiled := (obligations.coverage i).mp ⟨r, admitted⟩
      cases receipt : compiler i with
      | none => simp only [receipt, Option.isSome_none, Bool.false_eq_true] at compiled
      | some art =>
        exact (pipeline lowering backend execution (source_from_admitted i r admitted) art
          (obligations.artifact i r admitted w p art receipt)).fold.envOnto e allowed
    fixed := obligations.fixed
    session := obligations.session
    envAbs_agrees := fun _ => rfl
    ifixed := obligations.ifixed
    isession := obligations.isession
    creates_table := obligations.creates_table
    hashes_source := obligations.hashes_source
    refines := by
      intro i r admitted w p art receipt e allowed
      let passes := pipeline lowering backend execution (source_from_admitted i r admitted) art
        (obligations.artifact i r admitted w p art receipt)
      exact ⟨passes.fold.envSound e allowed, passes.fold.refines e allowed⟩
    decode_reach := obligations.decode_reach
    decode_commutes := obligations.decode_commutes }
  exact ⟨obligations.sourceApproved, obligations.sourceNonVacuous,
    obligations.targetApproved, obligations.keccak, publicProof, publicProof.transfer⟩

end Solidity.Roadmap.G49Final
